export type RiskDecision =

\| { allow: true; adjustedAmountUsd: number }

\| { allow: false; reason: string };

// \-\-\-\-\-\-\-\-\-- 主适配器 \-\-\-\-\-\-\-\-\--

export class ExecutionAdapter {

constructor(private env: Env) {}

/\*\*

\* 纯函数评估（无副作用，可单元测试）

\*/

evaluate(intent: ExecutionIntent, portfolio: PortfolioSnapshot):
RiskDecision {

// 1. 全局开关

if (this.env.EXECUTION_ENABLED !== \'true\') {

return { allow: false, reason: \'EXECUTION_ENABLED=false\' };

}

// 2. 日限额

const maxDaily = Number(this.env.MAX_DAILY_VOLUME_USDC ?? 50);

if (portfolio.dailyVolumeUsd \>= maxDaily) {

return { allow: false, reason: \`daily volume cap \${maxDaily} reached\`
};

}

// 3. 单笔限额

const maxPerTrade = Number(this.env.MAX_PER_TRADE_USDC ?? 5);

let amount = intent.suggestedAmountUsd ?? maxPerTrade \*
intent.strength;

amount = Math.min(amount, maxPerTrade);

// 4. 仓位占比

const maxPosPct = Number(this.env.MAX_POSITION_PCT ?? 10) / 100;

const currentPos = portfolio.positions.find(p =\> p.token ===
intent.token);

const currentValue = currentPos?.valueUsd ?? 0;

if (intent.side === \'buy\' && (currentValue + amount) /
portfolio.totalUsd \> maxPosPct) {

return { allow: false, reason: \'max position % exceeded\' };

}

// 5. 冷却时间

const cooldown = Number(this.env.TRADE_COOLDOWN_SECONDS ?? 300) \* 1000;

const last = portfolio.lastTradeAt\[intent.token\] ?? 0;

if (Date.now() - last \< cooldown) {

return { allow: false, reason: \'cooldown active\' };

}

// 6. 最低置信度

if (intent.confidence \< 0.45) {

return { allow: false, reason: \'confidence too low\' };

}

// 7. 可用资金

if (intent.side === \'buy\' && amount \> portfolio.availableUsd) {

return { allow: false, reason: \'insufficient available balance\' };

}

return { allow: true, adjustedAmountUsd: amount };

}

/\*\*

\* 执行入口（唯一允许产生链上交易的地方）

\*/

async execute(intent: ExecutionIntent): Promise\<ExecutionResult\> {

const portfolio = await this.getPortfolio();

const decision = this.evaluate(intent, portfolio);

if (!decision.allow) {

return {

status: \'rejected\',

intentId: intent.id,

reason: decision.reason,

timestamp: Date.now(),

};

}

// Shadow 模式（默认）

const realSpend = this.env.EXECUTION_REAL_SPEND === \'true\';

const shadow = this.env.EXECUTION_SHADOW !== \'false\'; // 默认 true

if (!realSpend \|\| shadow) {

// 只记录，不广播

await this.recordShadow(intent, decision.adjustedAmountUsd);

return {

status: \'shadow\',

intentId: intent.id,

amountIn: decision.adjustedAmountUsd.toFixed(4),

reason: realSpend ? \'shadow forced\' : \'REAL_SPEND=false\',

timestamp: Date.now(),

};

}

// 真正执行

try {

const result = await this.routeAndSwap(intent,
decision.adjustedAmountUsd);

await this.recordExecution(intent, result);

return result;

} catch (err: any) {

return {

status: \'failed\',

intentId: intent.id,

reason: err?.message ?? \'unknown error\',

timestamp: Date.now(),

};

}

}

// 让它们对 top 1-2 个信号投票

const candidates: Array\<{ flyId: string; arousal: number; behavior:
string }\> = \[\];

for (const \[flyId, drive\] of drives) {

const behavior = behaviors.get(flyId) \|\| \'REST\';

if (behavior === \'EXPLORE\' \|\| behavior === \'AGITATE\') {

candidates.push({

flyId,

arousal: drive.arousal ?? 0,

behavior,

});

}

}

// 按 arousal 排序，取前 3

candidates.sort((a, b) =\> b.arousal - a.arousal);

const topFlies = candidates.slice(0, 3);

if (topFlies.length === 0) return intents;

// 对每个高热信号生成一个意图（可调整策略）

for (const signal of topSignals.slice(0, 2)) {

const avgArousal = topFlies.reduce((s, f) =\> s + f.arousal, 0) /
topFlies.length;

const confidence = Math.min(0.95, signal.score \* 0.7 + avgArousal \*
0.3);

if (confidence \< 0.45) continue;

intents.push({

id: \`ext-\${Date.now()}-\${signal.token.slice(0, 6)}\`,

token: signal.token,

chain: signal.chain,

side: \'buy\', // 初期只做 buy，后续可加卖出逻辑

strength: avgArousal,

confidence,

maxSlippageBps: 150, // 1.5%

deadline: Math.floor(Date.now() / 1000) + 90,

sourceFlyIds: topFlies.map(f =\> f.flyId),

suggestedAmountUsd: Number(this.env.MAX_PER_TRADE_USDC \|\| 1) \*
avgArousal,

});

}

return intents;

}

\* 在原有 tick / settle 流程末尾调用

\*/

async processExternalIntents(

drives: Map\<string, any\>,

behaviors: Map\<string, string\>,

market: any

): Promise\<void\> {

const intents = this.buildExternalIntents(drives, behaviors, market);

if (intents.length === 0) return;

const adapter = new ExecutionAdapter(this.env);

for (const intent of intents) {

const result = await adapter.execute(intent);

\${result.txHash \|\| result.reason}\`);

} catch (err: any) {

console.error(\'\[EXTERNAL\] execute error\', intent.id, err?.message);

}

}

}

}
